compile stencils entry point to LLVM IR:
    mkdir -p output && clang++ -std=c++17 -O3 -fno-pic -fno-pie -mcmodel=medium -emit-llvm -c src/library/stencils_entry.cpp -o output/stencils_entry.bc

compile metavar compiler entry point to executable:
    mkdir -p output && clang++ -std=c++17 src/main.cpp src/library/stencil_registry.cpp src/utils/fake_symbol_resolver.cpp -o output/main \
        $(llvm-config --cxxflags --ldflags --system-libs --libs core orcjit irreader support executionengine native target) \
        -ldl -lpthread

both:
    mkdir -p output && clang++ -std=c++17 -O3 -fno-pic -fno-pie -mcmodel=medium -emit-llvm -c src/library/stencils_entry.cpp -o output/stencils_entry.bc && \
        clang++ -std=c++17 src/main.cpp src/library/stencil_registry.cpp src/utils/fake_symbol_resolver.cpp -o output/main \
        $(llvm-config --cxxflags --ldflags --system-libs --libs core orcjit irreader support executionengine native target) \
        -ldl -lpthread
